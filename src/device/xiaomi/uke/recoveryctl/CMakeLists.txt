cmake_minimum_required(VERSION 3.20)
project(uke_recoveryctl LANGUAGES CXX)
find_package(OpenSSL REQUIRED)
set(UKE_JSONCPP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../upstream/orangefox-android16/external/jsoncpp" CACHE PATH "Pinned Android JsonCpp source")
if(NOT EXISTS "${UKE_JSONCPP_SOURCE}/src/lib_json/json_reader.cpp")
    message(FATAL_ERROR "Set UKE_JSONCPP_SOURCE to the pinned JsonCpp checkout")
endif()
add_library(uke-jsoncpp STATIC
    ${UKE_JSONCPP_SOURCE}/src/lib_json/json_reader.cpp
    ${UKE_JSONCPP_SOURCE}/src/lib_json/json_value.cpp
    ${UKE_JSONCPP_SOURCE}/src/lib_json/json_writer.cpp)
target_include_directories(uke-jsoncpp SYSTEM PUBLIC ${UKE_JSONCPP_SOURCE}/include)
target_compile_definitions(uke-jsoncpp PRIVATE JSON_USE_EXCEPTION=0 JSONCPP_NO_LOCALE_SUPPORT)
add_library(uke-recovery STATIC libuke/core.cpp libuke/storage.cpp libuke/systems.cpp libuke/editor.cpp
    libuke/transaction.cpp libuke/backup.cpp libuke/block.cpp libuke/gpt.cpp libuke/stock.cpp libuke/partition.cpp libuke/display.cpp libuke/ownership.cpp libuke/diagnostics.cpp libuke/tools.cpp libuke/cli.cpp)
target_compile_features(uke-recovery PUBLIC cxx_std_20)
target_compile_options(uke-recovery PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
target_link_libraries(uke-recovery PUBLIC uke-jsoncpp OpenSSL::Crypto)
add_executable(uke-recoveryctl recoveryctl.cpp)
target_link_libraries(uke-recoveryctl PRIVATE uke-recovery)
target_compile_features(uke-recoveryctl PRIVATE cxx_std_20)
target_compile_options(uke-recoveryctl PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
