cmake_minimum_required(VERSION 3.20)
project(uke_recoveryctl LANGUAGES CXX)
find_package(OpenSSL REQUIRED)
set(UKE_JSONCPP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../upstream/orangefox-android16/external/jsoncpp" CACHE PATH "Pinned Android JsonCpp source")
if(NOT EXISTS "${UKE_JSONCPP_SOURCE}/src/lib_json/json_reader.cpp")
    message(FATAL_ERROR "Set UKE_JSONCPP_SOURCE to the pinned JsonCpp checkout")
endif()
add_library(uke-jsoncpp STATIC
    ${UKE_JSONCPP_SOURCE}/src/lib_json/json_reader.cpp
    ${UKE_JSONCPP_SOURCE}/src/lib_json/json_value.cpp
    ${UKE_JSONCPP_SOURCE}/src/lib_json/json_writer.cpp)
target_include_directories(uke-jsoncpp SYSTEM PUBLIC ${UKE_JSONCPP_SOURCE}/include)
target_compile_definitions(uke-jsoncpp PRIVATE JSON_USE_EXCEPTION=0 JSONCPP_NO_LOCALE_SUPPORT)
add_library(uke-recovery STATIC libuke/core.cpp libuke/storage.cpp libuke/systems.cpp libuke/editor.cpp
    libuke/transaction.cpp libuke/backup.cpp libuke/block.cpp libuke/gpt.cpp libuke/ownership.cpp libuke/diagnostics.cpp libuke/tools.cpp libuke/cli.cpp)
target_compile_features(uke-recovery PUBLIC cxx_std_20)
target_compile_options(uke-recovery PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
target_link_libraries(uke-recovery PUBLIC uke-jsoncpp OpenSSL::Crypto)
add_executable(uke-recoveryctl recoveryctl.cpp)
target_link_libraries(uke-recoveryctl PRIVATE uke-recovery)
target_compile_features(uke-recoveryctl PRIVATE cxx_std_20)
target_compile_options(uke-recoveryctl PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
include(CTest)
if(BUILD_TESTING)
    add_executable(uke-ure-tests ../../../../../tests/ure/test.cpp)
    target_include_directories(uke-ure-tests PRIVATE libuke)
    target_link_libraries(uke-ure-tests PRIVATE uke-recovery)
    target_compile_options(uke-ure-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    add_test(NAME ure-negative-and-transaction COMMAND uke-ure-tests)
    add_executable(uke-recovery-tests ../../../../../tests/ure/recovery.cpp)
    target_include_directories(uke-recovery-tests PRIVATE libuke)
    target_link_libraries(uke-recovery-tests PRIVATE uke-recovery)
    target_compile_options(uke-recovery-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    add_test(NAME ure-interruption-and-streaming COMMAND uke-recovery-tests)
    add_executable(uke-gpt-tests ../../../../../tests/ure/gpt.cpp)
    target_include_directories(uke-gpt-tests PRIVATE libuke)
    target_link_libraries(uke-gpt-tests PRIVATE uke-recovery)
    target_compile_options(uke-gpt-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    add_test(NAME ure-gpt-backup-repair-restore COMMAND uke-gpt-tests)
    add_executable(uke-ownership-tests ../../../../../tests/ure/ownership.cpp)
    target_include_directories(uke-ownership-tests PRIVATE libuke)
    target_link_libraries(uke-ownership-tests PRIVATE uke-recovery)
    target_compile_options(uke-ownership-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
    add_test(NAME ure-storage-ownership-policy COMMAND uke-ownership-tests)
endif()
